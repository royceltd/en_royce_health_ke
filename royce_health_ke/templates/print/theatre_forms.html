{#- Building blocks for the theatre forms (Consent for Operation, Theatre Checklist, Anaesthesia Record): ticked boxes and question/answer grids laid out like the hospital's paper forms. Used with health_document.html's styles. -#} {%- macro form_styles(ctx) -%} {%- endmacro -%} {#- A ticked or empty box: ticked for 1, "Yes", "Done", any answer except "No". -#} {%- macro box(value) -%} {% if value and value not in ("No", 0, "0") %}✓{% else %} {% endif %} {%- endmacro -%} {#- Question / answer rows: (label, value). Checks show a box; selects and text show their answer; unanswered rows leave a line to write on. -#} {%- macro qa(rows, checks=False) -%} {% for label, value in rows %} {% if not checks %} {% if value not in (None, "", 0) %}{% else %}{% endif %} {% endif %} {% endfor %}
{% if checks %}{{ box(value) }}{% endif %}{{ _(label) }}{{ value }}
{%- endmacro -%} {#- Rows split over two columns. -#} {%- macro qa2(rows, checks=False) -%} {% set half = ((rows | length) + 1) // 2 %}
{{ qa(rows[:half], checks) }} {{ qa(rows[half:], checks) }}
{%- endmacro -%} {%- macro part(number, title, note="") -%}
{{ number }}. {{ _(title) }}{% if note %} {{ note }}{% endif %}
{%- endmacro -%} {%- macro fill(value, width=140) -%} {% if value %}{{ value }}{% else %}{% endif %} {%- endmacro -%}
